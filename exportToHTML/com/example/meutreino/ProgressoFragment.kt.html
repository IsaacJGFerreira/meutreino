package com.example.meutreino

import android.app.AlertDialog
import android.content.Intent
import android.net.Uri
import android.os.Bundle
import androidx.fragment.app.Fragment
import android.view.LayoutInflater
import android.view.View
import android.view.ViewGroup
import android.widget.*
import androidx.activity.result.contract.ActivityResultContracts
import androidx.recyclerview.widget.RecyclerView
import androidx.recyclerview.widget.LinearLayoutManager
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

class ProgressoFragment : Fragment() {

    private lateinit var lista: MutableList<ProgressoRegistro>
    private lateinit var progressoAdapter: ProgressoAdapter

    private var uriFrente: String? = null
    private var uriLado: String? = null
    private var uriCostas: String? = null

    private enum class TipoFoto { FRENTE, LADO, COSTAS }
    private var tipoAtual: TipoFoto = TipoFoto.FRENTE

    private lateinit var tvFotosStatus: TextView

    // ✅ Seletor de imagem (galeria)
    private val pickImage = registerForActivityResult(ActivityResultContracts.OpenDocument()) { uri: Uri? ->
        if (uri == null) return@registerForActivityResult

        // ✅ manter permissão para reabrir depois
        val flags = Intent.FLAG_GRANT_READ_URI_PERMISSION or Intent.FLAG_GRANT_WRITE_URI_PERMISSION
        requireContext().contentResolver.takePersistableUriPermission(uri, flags)

        when (tipoAtual) {
            TipoFoto.FRENTE -> uriFrente = uri.toString()
            TipoFoto.LADO -> uriLado = uri.toString()
            TipoFoto.COSTAS -> uriCostas = uri.toString()
        }
        atualizarStatusFotos()
        Toast.makeText(requireContext(), "Foto selecionada!", Toast.LENGTH_SHORT).show()
    }

    override fun onCreateView(
        inflater: LayoutInflater, container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View {

        val view = inflater.inflate(R.layout.fragment_progresso, container, false)

        // 🔹 Campos / botões
        val etPeso = view.findViewById<EditText>(R.id.etPesoProgresso)
        tvFotosStatus = view.findViewById(R.id.tvFotosStatus)

        val btnFrente = view.findViewById<Button>(R.id.btnFotoFrente)
        val btnLado = view.findViewById<Button>(R.id.btnFotoLado)
        val btnCostas = view.findViewById<Button>(R.id.btnFotoCostas)
        val btnSalvar = view.findViewById<Button>(R.id.btnSalvarProgresso)

        // 🔹 RecyclerView
        val rv = view.findViewById<RecyclerView>(R.id.rvProgresso)
        rv.layoutManager = LinearLayoutManager(requireContext())

        // ✅ Carrega lista do celular
        lista = ProgressoRepository.carregar(requireContext()).toMutableList()

        // ✅ Adapter das miniaturas
        progressoAdapter = ProgressoAdapter(
            lista,
            onClick = { pos -> abrirComparacao(pos) },
            onLongClick = { pos -> confirmarApagar(pos) }
        )
        rv.adapter = progressoAdapter

        atualizarStatusFotos()

        // ✅ Selecionar fotos
        btnFrente.setOnClickListener {
            tipoAtual = TipoFoto.FRENTE
            pickImage.launch(arrayOf("image/*"))
        }
        btnLado.setOnClickListener {
            tipoAtual = TipoFoto.LADO
            pickImage.launch(arrayOf("image/*"))
        }
        btnCostas.setOnClickListener {
            tipoAtual = TipoFoto.COSTAS
            pickImage.launch(arrayOf("image/*"))
        }

        // ✅ Salvar progresso
        btnSalvar.setOnClickListener {
            val pesoStr = etPeso.text.toString().trim()
            val peso = pesoStr.toDoubleOrNull()

            if (peso == null || peso <= 0) {
                Toast.makeText(requireContext(), "Peso inválido.", Toast.LENGTH_SHORT).show()
                return@setOnClickListener
            }

            val data = SimpleDateFormat("dd/MM/yyyy", Locale.getDefault()).format(Date())

            val item = ProgressoRegistro(
                id = "${data}_${peso}",
                data = data,
                pesoKg = peso,
                fotoFrenteUri = uriFrente,
                fotoLadoUri = uriLado,
                fotoCostasUri = uriCostas
            )

            // ✅ Mais recente em cima
            lista.add(0, item)

            ProgressoRepository.salvar(requireContext(), lista)
            progressoAdapter.notifyDataSetChanged()

            // limpa para próxima semana
            etPeso.text.clear()
            uriFrente = null
            uriLado = null
            uriCostas = null
            atualizarStatusFotos()

            Toast.makeText(requireContext(), "Progresso salvo!", Toast.LENGTH_SHORT).show()
        }

        return view
    }

    // ✅ Apagar item (clique longo)
    private fun confirmarApagar(pos: Int) {
        AlertDialog.Builder(requireContext())
            .setTitle("Apagar progresso")
            .setMessage("Apagar registro de ${lista[pos].data}?")
            .setPositiveButton("Apagar") { _, _ ->
                lista.removeAt(pos)
                ProgressoRepository.salvar(requireContext(), lista)
                progressoAdapter.notifyDataSetChanged()
            }
            .setNegativeButton("Cancelar", null)
            .show()
    }

    // ✅ Comparação simples (peso atual vs anterior)
    private fun abrirComparacao(pos: Int) {
        val atual = lista[pos]
        val anterior = if (pos + 1 < lista.size) lista[pos + 1] else null

        val msg = if (anterior == null) {
            "Esse é o seu primeiro registro.\n\nPeso: ${atual.pesoKg} kg"
        } else {
            val dif = atual.pesoKg - anterior.pesoKg
            val sinal = if (dif >= 0) "+" else ""
            """
Atual: ${atual.data} — ${atual.pesoKg} kg
Anterior: ${anterior.data} — ${anterior.pesoKg} kg

Diferença: $sinal${String.format(Locale.getDefault(), "%.1f", dif)} kg
""".trimIndent()
        }

        AlertDialog.Builder(requireContext())
            .setTitle("Comparação semanal")
            .setMessage(msg)
            .setPositiveButton("OK", null)
            .show()
    }

    private fun atualizarStatusFotos() {
        val count = listOf(uriFrente, uriLado, uriCostas).count { !it.isNullOrBlank() }
        tvFotosStatus.text = "Fotos: $count/3"
    }
}
